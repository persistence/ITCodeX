# ITCodeX 元数据模块 - 数据模型设计

> 版本: v1.2
> 日期: 2026-09-05
> 内存模型与业务 Repository 以 `database/sql` 为准。
> 系统表（collections/fields/indexes/yaegi_scripts）走 GoFrame `dao` / `do` / `entity`，禁止手改生成代码。
> 边界见 [README · GoFrame 使用边界](./README.md#goframe-使用边界)。

## 0. 两套模型，不要混用

| 类型 | 位置 | 生成方式 | 用途 |
|------|------|----------|------|
| 系统表 entity / DO | `internal/model/entity`、`internal/model/do` | `gf gen dao`，禁止手改 | 系统表 Where/Data/Scan |
| 内存元数据模型 | `internal/model/metadata` 或 service 包内 | 手写 | `Database` / `Collection` / `Field` 运行时对象 |
| 业务行 | `map[string]interface{}` + `Record` | 手写 | 动态表没有稳定 schema，不能做 DO |

系统表写入示例：

```go
dao.Collections.Ctx(ctx).Data(do.Collections{
    Name:        name,
    DisplayName: displayName,
}).Insert()
```

不要对系统表使用 `g.Map`。不要为业务 Collection 执行 `gf gen dao`。

## 1. 核心数据结构（Go 结构体定义）

### 1.1 Database - 数据库管理器

```go
package metadata

import (
    "database/sql"
    "sync"

    "github.com/google/cel-go/cel"
)

// Database 元数据数据库管理器
type Database struct {
    mu sync.RWMutex

    // 业务表：database/sql。系统表优先 dao，需要时从 g.DB() 取 *sql.DB
    sqlDB *sql.DB

    // 数据表注册表: collectionName -> *Collection
    collections map[string]*Collection

    // 字段类型注册表: typeName -> FieldFactory
    fieldTypes map[string]FieldFactory

    // 自定义模型注册表
    models map[string]ModelFactory

    // 自定义Repository注册表
    repositories map[string]RepositoryFactory

    // 查询操作符注册表: opName -> OperatorFunc
    operators map[string]OperatorFunc

    // CEL环境缓存
    celEnvs map[string]*cel.Env

    // Yaegi脚本管理器
    yaegiManager *YaegiManager

    // 可选：生命周期回调（服务 CEL/Yaegi，非 NocoBase 事件总线）
    eventBus *EventBus

    // 配置选项
    options DatabaseOptions
}

type DatabaseOptions struct {
    // 表名前缀
    TablePrefix string

    // 是否开启日志
    Logging bool

    // MySQL DSN，例如：
    // root:123456@tcp(127.0.0.1:3306)/itcodex?parseTime=true&loc=Local
    DSN string

    // Yaegi脚本目录
    ScriptsPath string

    // 是否允许无 filter 清空业务表
    AllowTruncate bool
}
```

### 1.2 Collection - 数据表模型

```go
package metadata

import "sync"

// Collection 数据表定义
type Collection struct {
    mu sync.RWMutex

    // 所属数据库实例
    db *Database

    // 元数据ID
    ID int64 `json:"id"`

    // 表标识名（唯一，用于API引用）
    Name string `json:"name"`

    // 显示名称
    DisplayName string `json:"displayName"`

    // 数据库实际表名（可含前缀）
    TableName string `json:"tableName"`

    // 表类型: general/tree/calendar/comment/file
    Type CollectionType `json:"type"`

    // 描述
    Description string `json:"description"`

    // 分类标签
    Categories []string `json:"categories"`

    // 主键字段名，默认 "id"
    FilterTargetKey string `json:"filterTargetKey"`

    // 配置选项
    Options CollectionOptions `json:"options"`

    // 字段注册表: fieldName -> *Field
    fields map[string]*Field

    // 索引定义
    indexes []*Index

    // Repository实例（懒加载）
    repository Repository

    // 是否为系统表
    IsSystem bool `json:"isSystem"`

    // 是否自动生成ID
    AutoGenId bool `json:"autoGenId"`

    // 可排序字段配置
    Sortable *SortableConfig `json:"sortable,omitempty"`

    // 时间戳
    CreatedAt *gtime.Time `json:"createdAt"`
    UpdatedAt *gtime.Time `json:"updatedAt"`
    CreatedBy int64       `json:"createdBy"`
    UpdatedBy int64       `json:"updatedBy"`
}

type CollectionType string

const (
    CollectionTypeGeneral  CollectionType = "general"  // 一期实现
    CollectionTypeTree     CollectionType = "tree"     // 五期：邻接表，parent 字段见 TreeParentKey
    CollectionTypeCalendar CollectionType = "calendar" // 五期：起止时间字段
    CollectionTypeComment  CollectionType = "comment"  // 五期：挂到业务表外键
    CollectionTypeFile     CollectionType = "file"     // 五期：只存元数据，存储引擎另议
)

// 不做：view / sql / inherit / fdw（见设计方案 README 范围）

type CollectionOptions struct {
    // 简单分页模式（跳过count查询）
    SimplePagination bool `json:"simplePagination,omitempty"`

    // 树表父ID字段名
    TreeParentKey string `json:"treeParentKey,omitempty"`

    // 日历表开始时间字段
    CalendarStartField string `json:"calendarStartField,omitempty"`

    // 日历表结束时间字段
    CalendarEndField string `json:"calendarEndField,omitempty"`

    // 评论表关联外键
    CommentForeignKey string `json:"commentForeignKey,omitempty"`

    // 其他自定义选项
    Extra map[string]interface{} `json:"extra,omitempty"`
}

type SortableConfig struct {
    Name     string `json:"name"`
    ScopeKey string `json:"scopeKey,omitempty"`
}
```

### 1.3 Field - 字段模型

```go
package metadata

import "sync"

// Field 字段定义（抽象基类）
type Field interface {
    // 基础属性
    GetName() string
    GetType() string
    GetDataType() string
    GetCollection() *Collection

    // 配置方法
    Get(key string) interface{}
    Set(key string, value interface{})
    Merge(options map[string]interface{})

    // 数据库方法
    Bind() error
    Unbind() error
    Remove() error
    ExistsInDb() (bool, error)
    RemoveFromDb() error

    // CEL校验
    CompileCELPrograms() error
    Validate(value interface{}, data map[string]interface{}) error
}

// BaseField 字段基础实现
type BaseField struct {
    mu sync.RWMutex

    // 所属Collection
    collection *Collection

    // 元数据ID
    ID int64 `json:"id"`

    // 字段名
    Name string `json:"name"`

    // 显示名称
    DisplayName string `json:"displayName"`

    // 字段Interface类型
    Type string `json:"type"`

    // 数据库存储类型
    DataType string `json:"dataType"`

    // 界面类型
    InterfaceType string `json:"interfaceType"`

    // 是否系统字段
    IsSystem bool `json:"isSystem"`

    // 是否必填
    IsRequired bool `json:"isRequired"`

    // 是否唯一
    IsUnique bool `json:"isUnique"`

    // 默认值
    DefaultValue interface{} `json:"defaultValue"`

    // 描述
    Description string `json:"description"`

    // 排序
    Sort int `json:"sort"`

    // 字段配置
    Options map[string]interface{} `json:"options"`

    // CEL校验规则（编译后）
    celPrograms []cel.Program

    // 校验规则配置
    ValidationRules []*CELValidationRule `json:"validationRules"`

    // 时间戳
    CreatedAt *gtime.Time `json:"createdAt"`
    UpdatedAt *gtime.Time `json:"updatedAt"`
}

// CELValidationRule CEL校验规则
type CELValidationRule struct {
    // 规则名称（用于错误提示）
    Name string `json:"name"`

    // CEL表达式，必须返回bool或string（错误消息）
    Expression string `json:"expression"`

    // 错误消息
    ErrorMessage string `json:"errorMessage"`

    // 涉及的字段列表（用于确定是否需要触发）
    Dependencies []string `json:"dependencies"`
}

// FieldFactory 字段类型工厂函数
type FieldFactory func(collection *Collection, options map[string]interface{}) (Field, error)
```

### 1.5 Repository - 数据仓库接口

```go
package metadata

import (
    "context"
    "database/sql"
)

// Repository 业务表数据仓库。不使用 dao/do。
type Repository interface {
    Collection() *Collection

    Find(ctx context.Context, options *FindOptions) ([]Record, error)
    FindOne(ctx context.Context, options *FindOneOptions) (Record, error)
    FindAndCount(ctx context.Context, options *FindOptions) ([]Record, int, error)
    Count(ctx context.Context, options *CountOptions) (int, error)

    Create(ctx context.Context, options *CreateOptions) (Record, error)
    CreateMany(ctx context.Context, options *CreateManyOptions) ([]Record, error)
    Update(ctx context.Context, options *UpdateOptions) (Record, error)
    Destroy(ctx context.Context, options *DestroyOptions) (int, error)

    Query(ctx context.Context, sqlText string, args ...interface{}) ([]Record, error)
}

// Record 数据记录接口
type Record interface {
    Get(key string) interface{}
    Set(key string, value interface{})
    Data() map[string]interface{}
    Id() interface{}
    Save(ctx context.Context) error
    Delete(ctx context.Context) error
}

// ---- 查询选项 ----

type Filter map[string]interface{}
type Sort []string
type Fields []string
type Appends []string

type CommonOptions struct {
    Ctx context.Context
    Tx  *sql.Tx

    Filter  Filter  `json:"filter,omitempty"`
    Fields  Fields  `json:"fields,omitempty"`
    Except  Fields  `json:"except,omitempty"`
    Appends Appends `json:"appends,omitempty"`
    Sort    Sort    `json:"sort,omitempty"`
}

type FindOptions struct {
    CommonOptions
    FilterByTk interface{} `json:"filterByTk,omitempty"`
    Limit      int         `json:"limit,omitempty"`
    Offset     int         `json:"offset,omitempty"`
    Page       int         `json:"page,omitempty"`
    PageSize   int         `json:"pageSize,omitempty"`
}

type FindOneOptions struct {
    CommonOptions
    FilterByTk interface{} `json:"filterByTk,omitempty"`
}

type CountOptions struct {
    CommonOptions
}

type CreateOptions struct {
    CommonOptions
    Values      map[string]interface{}   `json:"values"`
    Whitelist   []string                 `json:"whitelist,omitempty"`
    Blacklist   []string                 `json:"blacklist,omitempty"`
}

type CreateManyOptions struct {
    CommonOptions
    Records   []map[string]interface{} `json:"records"`
    Whitelist []string                 `json:"whitelist,omitempty"`
    Blacklist []string                 `json:"blacklist,omitempty"`
}

type UpdateOptions struct {
    CommonOptions
    FilterByTk interface{}              `json:"filterByTk,omitempty"`
    Values     map[string]interface{}   `json:"values"`
    Whitelist  []string                 `json:"whitelist,omitempty"`
    Blacklist  []string                 `json:"blacklist,omitempty"`
}

type DestroyOptions struct {
    CommonOptions
    FilterByTk interface{} `json:"filterByTk,omitempty"`
    Truncate   bool       `json:"truncate,omitempty"`
}
```

## 2. 内置字段类型定义

### 2.1 基础字段（Basic Fields）

```go
// 单行文本
type StringField struct {
    *BaseField
}
// 数据类型: varchar(255)/text（可配置）
// 支持校验: required/minLen/maxLen/pattern/email/url/uuid

// 多行文本
type TextField struct {
    *BaseField
}
// 数据类型: text/longtext
// 支持校验: required/minLen/maxLen

// 手机号
type PhoneField struct {
    *BaseField
}
// 数据类型: varchar(50)
// 支持校验: required/phoneFormat

// 邮箱
type EmailField struct {
    *BaseField
}
// 数据类型: varchar(255)
// 支持校验: required/emailFormat

// URL
type UrlField struct {
    *BaseField
}
// 数据类型: varchar(2048)
// 支持校验: required/urlFormat

// 数字
type NumberField struct {
    *BaseField
}
// 数据类型: double/float/decimal（可配置精度、小数位）
// 支持校验: required/min/max/exclusiveMin/exclusiveMax/multipleOf

// 整数
type IntegerField struct {
    *BaseField
}
// 数据类型: integer/bigint
// 支持校验: required/min/max/integer

// 百分比
type PercentField struct {
    *BaseField
}
// 数据类型: double（0-100）
// 支持校验: required/min/max

// 颜色
type ColorField struct {
    *BaseField
}
// 数据类型: varchar(20)（十六进制颜色值）
// 支持校验: required/colorFormat

// 图标
type IconField struct {
    *BaseField
}
// 数据类型: varchar(100)

// 密码
type PasswordField struct {
    *BaseField
}
// 数据类型: varchar(255)（存储scrypt/argon2哈希值）
// 支持校验: required/minLen/maxLen/pattern
```

### 2.5 关系字段（Relation Fields）

```go
// 多对一
type BelongsToField struct {
    *BaseField
    Target      string // 目标表名
    ForeignKey  string // 外键字段名
    SourceKey   string // 目标表主键
}
// 数据类型: 外键字段类型（通常bigint）

// 一对多
type HasManyField struct {
    *BaseField
    Target      string
    ForeignKey  string
    SourceKey   string
}
// 虚拟字段，不直接存储在本表

// 一对一
type HasOneField struct {
    *BaseField
    Target      string
    ForeignKey  string
    SourceKey   string
}
// 虚拟字段

// 多对多
type BelongsToManyField struct {
    *BaseField
    Target      string
    Through     string // 中间表名
    ForeignKey  string
    OtherKey    string
    SourceKey   string
    TargetKey   string
}
// 虚拟字段，通过中间表关联

// 多对多（数组）
type BelongsToManyArrayField struct {
    *BaseField
    Target      string
    TargetKey   string
}
// 数据类型: json（ID数组）
```

### 2.8 系统信息字段（System Fields）

业务表的 `createdAt` / `updatedAt` 由 Repository 写入（`database/sql`），**不是** GoFrame ORM 自动维护。ORM 自动时间只作用于系统表。

```go
// 创建时间
type CreatedAtField struct {
    *BaseField
}
// 数据类型: datetime，自动填充

// 最后更新时间
type UpdatedAtField struct {
    *BaseField
}
// 数据类型: datetime，自动填充

// 创建人
type CreatedByField struct {
    *BaseField
}
// 数据类型: bigint，自动填充

// 最后更新人
type UpdatedByField struct {
    *BaseField
}
// 数据类型: bigint，自动填充

// 表OID
type TableOIDField struct {
    *BaseField
}
// 数据类型: varchar(100)，唯一标识
```

## 3. 字段类型到 MySQL 数据类型映射

| 字段类型 | MySQL 数据类型 | Go 类型 |
|----------|----------------|---------|
| string | VARCHAR(255) / TEXT | string |
| text | TEXT / LONGTEXT | string |
| integer | INT | int32 |
| bigInt | BIGINT | int64 |
| float | FLOAT | float32 |
| double | DOUBLE | float64 |
| decimal | DECIMAL(precision,scale) | float64 / string |
| boolean | TINYINT(1) / BOOLEAN | bool |
| date | DATE | time.Time / string |
| time | TIME | time.Time / string |
| datetime | DATETIME | time.Time |
| timestamp | TIMESTAMP / BIGINT | time.Time |
| json | JSON | []byte / string / map |
| uuid | CHAR(36) | string |
| blob | BLOB / LONGBLOB | []byte |
| point | JSON (GeoJSON) | []byte |
| password | VARCHAR(255) | string (hash) |
