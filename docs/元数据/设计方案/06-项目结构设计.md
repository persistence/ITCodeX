## 2. GoFrame 路由注册

### 2.1 路由初始化

```go
// main.go (位于 d:\code\ITCodeX\server\main.go)
package main

import (
    "github.com/gogf/gf/v2/frame/g"
    "github.com/gogf/gf/v2/net/ghttp"

    "itcodex/server/controller/metadata"
    "itcodex/server/logic/metadata"
    mdMiddleware "itcodex/server/middleware"
)

func main() {
    s := g.Server()

    // 初始化元数据模块
    metaSvc := metadata.NewDatabase(metadata.DatabaseOptions{
        StoragePath: "./data/metadata.db", // CobaltDB文件路径
        ScriptsPath: "./yaegi/scripts",
        TablePrefix: "",
        Logging:     true,
    })
    if err := metaSvc.Bootstrap(); err != nil {
        g.Log().Fatal(ctx, "元数据模块初始化失败:", err)
    }

    // 注册中间件
    s.Use(mdMiddleware.MetadataContext(metaSvc))

    // API路由组
    s.Group("/api", func(group *ghttp.RouterGroup) {
        // 元数据管理接口（需要鉴权，此处省略）
        metaGroup := group.Group("/meta")
        metaGroup.Bind(
            metadata.NewMetaController(metaSvc),
        )

        // 自定义API路由（Yaegi）
        customGroup := group.Group("/custom")
        mdMiddleware.RegisterYaegiCustomRoutes(metaSvc, customGroup)
    })

    // 动态CRUD路由（通过中间件处理）
    crudGroup := s.Group("/api/c")
    crudGroup.Use(mdMiddleware.YaegiCRUDHook(metaSvc))
    crudGroup.ALL("/*", metadata.NewCRUDController(metaSvc))

    s.SetPort(8000)
    s.Run()
}
```

### 2.2 动态路由处理

由于 Collection 是动态创建的，无法静态注册路由，因此使用通配路由配合中间件：

```go
// middleware/yaegi_custom_route.go (位于 d:\code\ITCodeX\server\middleware\yaegi_custom_route.go)
package middleware

import (
    "context"
    "strings"

    "github.com/gogf/gf/v2/frame/g"
    "github.com/gogf/gf/v2/net/ghttp"

    "itcodex/server/logic/metadata"
)

// MetadataContext 注入元数据服务到请求上下文
func MetadataContext(db *metadata.Database) func(r *ghttp.Request) {
    return func(r *ghttp.Request) {
        ctx := r.Context()
        ctx = context.WithValue(ctx, "metadata_db", db)
        r.SetCtx(ctx)
        r.Middleware.Next()
    }
}

func GetMetadataDBFromCtx(ctx context.Context) *metadata.Database {
    return ctx.Value("metadata_db").(*metadata.Database)
}

// RegisterYaegiCustomRoutes 注册Yaegi自定义API路由
func RegisterYaegiCustomRoutes(db *metadata.Database, group *ghttp.RouterGroup) {
    // 表级自定义API: /custom/c/:collection/*
    group.ALL("/c/:collectionName/*", func(r *ghttp.Request) {
        collectionName := r.GetRouterString("collectionName")
        subPath := r.GetRouterString("*")

        db := GetMetadataDBFromCtx(r.Context())
        if !db.HasCollection(collectionName) {
            r.Response.WriteStatus(404)
            r.Response.WriteJson(g.Map{"code": 404, "message": "数据表不存在"})
            return
        }

        // 查找对应的Yaegi脚本并执行
        fullPath := "/c/" + collectionName + "/" + strings.TrimPrefix(subPath, "/")
        script := db.Yaegi().FindCustomAPI(r.Method, fullPath)
        if script == nil {
            r.Response.WriteStatus(404)
            r.Response.WriteJson(g.Map{"code": 404, "message": "自定义API不存在"})
            return
        }

        // 构建HTTP上下文并执行脚本
        yaegiCtx := buildYaegiContext(r)
        if err := db.Yaegi().ExecuteCustomAPI(script, yaegiCtx); err != nil {
            r.Response.WriteStatus(500)
            r.Response.WriteJson(g.Map{"code": 500, "message": err.Error()})
            return
        }
    })

    // 全局自定义API: /custom/global/*
    group.ALL("/global/*", func(r *ghttp.Request) {
        subPath := r.GetRouterString("*")
        fullPath := "/global/" + strings.TrimPrefix(subPath, "/")

        db := GetMetadataDBFromCtx(r.Context())
        script := db.Yaegi().FindCustomAPI(r.Method, fullPath)
        if script == nil {
            r.Response.WriteStatus(404)
            r.Response.WriteJson(g.Map{"code": 404, "message": "自定义API不存在"})
            return
        }

        yaegiCtx := buildYaegiContext(r)
        if err := db.Yaegi().ExecuteCustomAPI(script, yaegiCtx); err != nil {
            r.Response.WriteStatus(500)
            r.Response.WriteJson(g.Map{"code": 500, "message": err.Error()})
            return
        }
    })
}

func buildYaegiContext(r *ghttp.Request) *yaegictx.YaegiHTTPContext {
    params := make(map[string]string)
    r.GetRouterMap(params)

    query := make(map[string]string)
    r.GetQueryMap(query)

    return &yaegictx.YaegiHTTPContext{
        Params:   params,
        Query:    query,
        Headers:  r.Header,
        Request:  &yaegictx.YaegiRequest{Raw: r},
        Response: &yaegictx.YaegiResponse{Raw: r},
        User:     getCurrentUser(r), // 从认证中间件获取
    }
}
```

```go
// logic/metadata/database.go (位于 d:\code\ITCodeX\server\logic\metadata\database.go)
package metadata

func NewDatabase(options DatabaseOptions) *Database {
    db := &Database{
        collections:    make(map[string]*Collection),
        fieldTypes:     make(map[string]FieldFactory),
        repositories:   make(map[string]RepositoryFactory),
        operators:      make(map[string]OperatorFunc),
        celEnvs:        make(map[string]*cel.Env),
        options:        options,
    }
    db.eventBus = NewEventBus()
    return db
}

// Bootstrap 启动元数据模块
func (db *Database) Bootstrap() error {
    // 1. 初始化CobaltDB连接（通过GoFrame ORM适配层）
    if err := db.initCobaltDB(); err != nil {
        return err
    }

    // 2. 同步系统元数据表（collections/fields/indexes/yaegi_scripts）
    if err := db.syncSystemTables(); err != nil {
        return err
    }

    // 3. 注册内置字段类型
    db.registerBuiltinFieldTypes()

    // 4. 注册内置查询操作符
    db.registerBuiltinOperators()

    // 5. 从数据库加载Collection和Field定义
    if err := db.loadMetadataFromDB(); err != nil {
        return err
    }

    // 6. 同步业务表结构到数据库
    if err := db.syncBusinessTables(); err != nil {
        return err
    }

    // 7. 初始化Yaegi脚本引擎
    yaegiMgr, err := NewYaegiManager(db, db.options.ScriptsPath)
    if err != nil {
        return err
    }
    db.yaegiManager = yaegiMgr

    // 8. 加载并编译Yaegi脚本
    if err := db.loadYaegiScripts(); err != nil {
        return err
    }

    return nil
}

// initCobaltDB 初始化CobaltDB连接
func (db *Database) initCobaltDB() error {
    // CobaltDB作为嵌入式数据库，使用GoFrame的适配
    // 配置示例：
    // [database]
    //     type = "sqlite"  # CobaltDB兼容SQLite协议
    //     link = "cobaltdb::@file(./data/metadata.db)"
    // 或者使用GoFrame的数据库配置动态添加
    conn, err := gdb.New(gdb.ConfigNode{
        Type: "sqlite", // 假设CobaltDB通过SQLite接口兼容
        Link: "cobaltdb::@file(" + db.options.StoragePath + ")",
        // 根据CobaltDB实际GoFrame驱动配置
    })
    if err != nil {
        return err
    }
    db.gdb = conn
    return nil
}
```

### 3.2 Collection CRUD 逻辑（元数据管理）

```go
// logic/metadata/collection.go (位于 d:\code\ITCodeX\server\logic\metadata\collection.go)
package metadata

import (
    "context"
    "fmt"

    "github.com/gogf/gf/v2/os/gtime"
)

// CreateCollection 创建数据表
func (db *Database) CreateCollection(ctx context.Context, input CreateCollectionInput) (*Collection, error) {
    // 1. 验证name唯一性
    if db.HasCollection(input.Name) {
        return nil, fmt.Errorf("表名 %s 已存在", input.Name)
    }

    // 2. 开启事务
    tx, err := db.gdb.Ctx(ctx).Begin()
    if err != nil {
        return nil, err
    }
    defer tx.Rollback()

    // 3. 创建Collection对象
    now := gtime.Now()
    collection := &Collection{
        Name:           input.Name,
        DisplayName:    input.DisplayName,
        TableName:      db.options.TablePrefix + input.Name,
        Type:           input.Type,
        Description:    input.Description,
        Categories:     input.Categories,
        Options:        input.Options,
        FilterTargetKey: "id",
        AutoGenId:      true,
        fields:         make(map[string]*Field),
        db:             db,
        CreatedAt:      now,
        UpdatedAt:      now,
        CreatedBy:      0, // 从ctx获取用户ID
        UpdatedBy:      0,
    }

    // 4. 添加预设字段（id/createdAt/createdBy/updatedAt/updatedBy）
    if len(input.PresetFields) > 0 {
        if err := collection.addPresetFields(input.PresetFields); err != nil {
            return nil, err
        }
    }

    // 5. 添加用户定义的字段
    for _, fieldInput := range input.Fields {
        if err := collection.AddField(ctx, fieldInput); err != nil {
            return nil, err
        }
    }

    // 6. 添加索引
    for _, idxInput := range input.Indexes {
        collection.AddIndex(idxInput)
    }

    // 7. 保存Collection元数据到系统表
    collectionID, err := collection.saveToDB(ctx, tx)
    if err != nil {
        return nil, err
    }
    collection.ID = collectionID

    // 8. 保存字段和索引元数据
    if err := collection.saveFieldsToDB(ctx, tx); err != nil {
        return nil, err
    }

    // 9. 在数据库中创建实际表结构
    if err := collection.syncToDBCreate(ctx, tx); err != nil {
        return nil, err
    }

    // 10. 注册到内存
    db.mu.Lock()
    db.collections[collection.Name] = collection
    db.mu.Unlock()

    // 11. 编译CEL校验规则
    if err := collection.CompileCELRules(); err != nil {
        return nil, err
    }

    // 12. 触发事件
    db.eventBus.Emit("afterDefineCollection", collection)

    // 13. 提交事务
    if err := tx.Commit(); err != nil {
        return nil, err
    }

    return collection, nil
}

// GetCollection 获取数据表
func (db *Database) GetCollection(name string) *Collection {
    db.mu.RLock()
    defer db.mu.RUnlock()
    return db.collections[name]
}

func (db *Database) HasCollection(name string) bool {
    db.mu.RLock()
    defer db.mu.RUnlock()
    _, ok := db.collections[name]
    return ok
}
```

```go
// controller/metadata/crud_controller.go (位于 d:\code\ITCodeX\server\controller\metadata\crud_controller.go)
package metadata

import (
    "strings"

    "github.com/gogf/gf/v2/frame/g"
    "github.com/gogf/gf/v2/net/ghttp"

    metaLogic "itcodex/server/logic/metadata"
)

type CRUDController struct {
    db *metaLogic.Database
}

func NewCRUDController(db *metaLogic.Database) *CRUDController {
    return &CRUDController{db: db}
}

// 通配路由入口
func (c *CRUDController) Handle(r *ghttp.Request) {
    ctx := r.Context()
    path := r.URL.Path
    // 去掉 /api/c/ 前缀
    path = strings.TrimPrefix(path, "/api/c/")
    parts := strings.Split(path, "/")

    if len(parts) < 1 || parts[0] == "" {
        r.Response.WriteStatus(404)
        return
    }

    collectionName := parts[0]
    collection := c.db.GetCollection(collectionName)
    if collection == nil {
        r.Response.WriteJson(g.Map{
            "code":    404,
            "message": "数据表不存在: " + collectionName,
        })
        return
    }

    repo := collection.Repository()

    // 根据HTTP方法和路径分发
    method := r.Method
    switch {
    case method == "GET" && len(parts) == 1:
        // GET /api/c/:collection -> 列表查询
        c.handleFind(r, repo)
    case method == "GET" && len(parts) == 2 && parts[1] == "count":
        // GET /api/c/:collection/count -> 统计
        c.handleCount(r, repo)
    case method == "GET" && len(parts) == 2:
        // GET /api/c/:collection/:id -> 查询单条
        c.handleFindOne(r, repo, parts[1])
    case method == "POST" && len(parts) == 1:
        // POST /api/c/:collection -> 创建
        c.handleCreate(r, repo)
    case method == "POST" && len(parts) == 2 && parts[1] == "batch":
        // POST /api/c/:collection/batch -> 批量创建
        c.handleCreateMany(r, repo)
    case method == "PUT" && len(parts) == 2:
        // PUT /api/c/:collection/:id -> 更新
        c.handleUpdate(r, repo, parts[1])
    case method == "PUT" && len(parts) == 1:
        // PUT /api/c/:collection -> 批量更新
        c.handleUpdateMany(r, repo)
    case method == "DELETE" && len(parts) == 2:
        // DELETE /api/c/:collection/:id -> 删除
        c.handleDestroy(r, repo, parts[1])
    case method == "DELETE" && len(parts) == 1:
        // DELETE /api/c/:collection -> 批量删除/清空
        c.handleDestroyMany(r, repo)
    default:
        r.Response.WriteStatus(404)
    }
}
```

```
   ├─ 同步系统表

   ├─ 注册字段类型和操作符
   ├─ 同步业务表
   └─ 加载Yaegi脚本
5. 注册中间件（认证、元数据上下文、Yaegi钩子）
6. 注册路由
7. 启动HTTP服务
```

## 5. 关键开发优先级

1. **第一阶段**：Database核心 + 基础字段类型（string/number/bool/date）+ Repository基础CRUD + 简单Filter
2. **第二阶段**：完整字段类型 + CEL校验（单字段）+ 元数据管理API
3. **第三阶段**：CEL多字段校验 + 关系字段 + 关联查询
4. **第四阶段**：Yaegi钩子 + 自定义API + 脚本管理
5. **第五阶段**：高级字段类型（公式/加密/几何等）+ 性能优化
